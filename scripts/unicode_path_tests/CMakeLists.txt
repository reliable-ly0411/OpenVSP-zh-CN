cmake_minimum_required(VERSION 3.24)
project(UnicodePathRegression LANGUAGES CXX)
add_executable(unicode_path_probe probe.cpp)
target_include_directories(unicode_path_probe PRIVATE ../../src/util)
target_compile_features(unicode_path_probe PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(unicode_path_probe PRIVATE /utf-8)
endif()

# Compile the actual source functions, not a hand-written approximation. The
# immutable baseline is needed only for this read-only causal comparison.
set(PATTERN_BASELINE 1d6b0b6078716db5854ff04a92610aa58956a0c6)
execute_process(COMMAND git show "${PATTERN_BASELINE}:src/util/FileUtil.cpp"
  WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../.."
  OUTPUT_VARIABLE BASELINE_FILEUTIL RESULT_VARIABLE BASELINE_STATUS)
if(NOT BASELINE_STATUS EQUAL 0)
  message(FATAL_ERROR "Pinned FileUtil baseline is unavailable; fetch its history")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../src/util/FileUtil.cpp" PATCHED_FILEUTIL)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/../../src/util/FileUtil.cpp")
set(PATTERN_SOURCE "#include \"FileUtil.h\"\n#include <algorithm>\n#include <regex>\n")
foreach(VARIANT IN ITEMS BASELINE PATCHED)
  set(SOURCE "${${VARIANT}_FILEUTIL}")
  string(FIND "${SOURCE}" "void GetPathFile(" START)
  string(FIND "${SOURCE}" "void AppendFile_BtoA(" END)
  if(START LESS 0 OR END LESS START)
    message(FATAL_ERROR "Cannot locate actual GetPathFile source")
  endif()
  math(EXPR LENGTH "${END} - ${START}")
  string(SUBSTRING "${SOURCE}" ${START} ${LENGTH} PATH_FUNCTION)
  string(FIND "${SOURCE}" "std::vector< fs::path > get_files_matching_pattern(" START)
  string(FIND "${SOURCE}" "void remove_files(" END)
  if(START LESS 0 OR END LESS START)
    message(FATAL_ERROR "Cannot locate actual pattern-matching source")
  endif()
  math(EXPR LENGTH "${END} - ${START}")
  string(SUBSTRING "${SOURCE}" ${START} ${LENGTH} MATCH_FUNCTIONS)
  string(TOLOWER "${VARIANT}" NS)
  string(SHA256 FUNCTION_SHA "${PATH_FUNCTION}${MATCH_FUNCTIONS}")
  string(APPEND PATTERN_SOURCE "namespace ${NS} {\nstd::vector<fs::path> get_files_matching_pattern(const fs::path&, const std::string&);\n${PATH_FUNCTION}${MATCH_FUNCTIONS}\nconst char* source_sha256() { return \"${FUNCTION_SHA}\"; }\n}\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/pattern_functions.cpp" "${PATTERN_SOURCE}")
add_executable(unicode_pattern_probe pattern_probe.cpp "${CMAKE_CURRENT_BINARY_DIR}/pattern_functions.cpp")
target_include_directories(unicode_pattern_probe PRIVATE ../../src/util)
target_compile_features(unicode_pattern_probe PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(unicode_pattern_probe PRIVATE /utf-8)
endif()
